    ocf = _choose(cash, "NETCASH_OPERATE")

"""A 股公告正文：只按东财 art_code 取正文，失败保留标题事实。"""
import logging
import re
from typing import Any

import requests

logger = logging.getLogger(__name__)
CONTENT_URL = "https://np-cnotice-stock.eastmoney.com/api/content/ann"


def extract_art_code(url: str) -> str | None:
    match = re.search(r"(AN\d{10,})", str(url or ""))
    return match[1] if match else None


def fetch_notice_content(art_code: str) -> str:
    response = requests.get(
        CONTENT_URL,
        params={"art_code": art_code, "client_source": "web", "page_index": 1},
        timeout=8,
    )
    response.raise_for_status()
    payload: Any = response.json()
    data = payload.get("data") if isinstance(payload, dict) else None
    content = data.get("notice_content") if isinstance(data, dict) else None
    if not isinstance(content, str) or not content.strip():
        raise ValueError("公告正文为空")
    return re.sub(r"\s+", " ", content).strip()
```

def mark_related(items: list[dict[str, Any]], *, name: str, symbol: str) -> list[dict[str, Any]]:
    """无模型时只做名称/代码命中标记，不伪造情绪或可评估性。"""
    from data.news_relevance import name_aliases

    aliases = name_aliases(name, symbol)
    output = []
    for item in items:
        text = f"{item.get('title', '')} {item.get('content', '')}"
        related = bool(aliases) and any(alias in text for alias in aliases)
        output.append({**item, "related": related, "assessable": False})
    return output
```

def fetch_market_caps(symbols: list[str], *, timeout: float = 10) -> dict[str, float]:
    """同一批 ulist.np 取目标与同行当前市值；单位与东财一致。"""
    from data.hk_eastmoney import finite_number
    output: dict[str, float] = {}
    for offset in range(0, len(symbols), 50):
        batch = symbols[offset:offset + 50]
        secids = ",".join(f"116.{s.removesuffix('.HK')}" for s in batch)
        response = requests.get(
            "https://push2.eastmoney.com/api/qt/ulist.np/get",
            params={"secids": secids, "fields": "f12,f20", "fltt": 2, "invt": 2},
            timeout=timeout,
        )
        response.raise_for_status()
        rows = (response.json().get("data") or {}).get("diff") or []
        by_code = {f"{row.get('f12')}.HK": finite_number(row.get("f20")) for row in rows}
        for symbol in batch:
            value = by_code.get(symbol)
            if value is not None:
                output[symbol] = value
    return output
```

def verify_balance(period: date, balance_rows: list[dict[str, Any]],
                   indicator: dict[str, Any] | None) -> tuple[bool, str, str]:
    """用指标接口的总资产/归母权益证明资产负债表币种；缺一或不等即拒绝。"""
    if indicator is None or _date(indicator.get("REPORT_DATE")) != period:
        return False, "", "分析指标与资产负债表报告期不一致"
    currency = FinancialData.normalize_report_currency(indicator.get("CURRENCY"))
    if not currency:
        return False, "", "分析指标未提供有效币种"
    amounts = {str(row.get("STD_ITEM_CODE")): finite_number(row.get("AMOUNT"))
               for row in balance_rows if _date(row.get("REPORT_DATE")) == period}
    assets, equity = amounts.get("004009999"), amounts.get("004030999")
    if assets is None or equity is None:
        return False, currency, "资产负债表核验科目缺失"
    if not (_close(assets, indicator.get("TOTAL_ASSETS")) and _close(equity, indicator.get("TOTAL_PARENT_EQUITY"))):
        return False, currency, "资产负债表金额与指标接口不一致"
    return True, currency, ""
```

def dedupe_pending(items: list[str]) -> list[str]:

    seen: set[str] = set()

        if clauses:
