```python
"""A 股东财日频估值历史；只返回可核实字段，不做跨源拼接。"""
import logging
from dataclasses import dataclass
from datetime import date, datetime
from typing import Any

import akshare as ak

from data.collection_state import ValueResult, make_record
from utils.numbers import is_finite_number

logger = logging.getLogger(__name__)
SOURCE = "东方财富日频估值历史"
_REQUIRED = {"数据日期", "PE(TTM)", "市净率", "市销率", "总市值", "总股本"}


@dataclass(frozen=True)
class ValuationHistoryRow:
    trade_date: date
    close: float | None
    pe_ttm: float | None
    pb: float | None
    ps_ttm: float | None
    market_cap: float | None
    total_shares: float | None


def _date(value: Any) -> date | None:
    if isinstance(value, datetime):
        return value.date()
    if isinstance(value, date):
        return value
    try:
        return date.fromisoformat(str(value)[:10])
    except (TypeError, ValueError):
        return None


def _number(value: Any) -> float | None:
    if isinstance(value, bool):
        return None
    try:
        number = float(value)
    except (TypeError, ValueError):
        return None
    return number if is_finite_number(number) else None


def parse_history(frame: Any, today: date | None = None) -> list[ValuationHistoryRow]:
    if frame is None or getattr(frame, "empty", True):
        raise ValueError("东财估值历史为空")
    missing = _REQUIRED - set(frame.columns)
    if missing:
        raise ValueError(f"东财估值历史缺少列：{', '.join(sorted(missing))}")
    cutoff = today or datetime.now().astimezone().date()
    rows: dict[date, ValuationHistoryRow] = {}
    conflicts: set[date] = set()
    for record in frame.to_dict("records"):
        day = _date(record.get("数据日期"))
        if day is None or day > cutoff:
            continue
        row = ValuationHistoryRow(
            trade_date=day,
            close=_number(record.get("当日收盘价")),
            pe_ttm=_number(record.get("PE(TTM)")),
            pb=_number(record.get("市净率")),
            ps_ttm=_number(record.get("市销率")),
            market_cap=_number(record.get("总市值")),
            total_shares=_number(record.get("总股本")),
        )
        previous = rows.get(day)
        if previous is not None and previous != row:
            conflicts.add(day)
            continue
        rows[day] = row
    for day in conflicts:
        rows.pop(day, None)
    if not rows:
        raise ValueError("东财估值历史没有有效日期")
    return [rows[key] for key in sorted(rows)]


def load_history(symbol: str) -> ValueResult[list[ValuationHistoryRow]]:
    if len(symbol) != 6 or not symbol.isdigit():
        return ValueResult(None, make_record(
            SOURCE, "failed", error_kind="invalid_symbol",
            error_message="A 股代码必须为六位数字"))
    try:
        rows = parse_history(ak.stock_value_em(symbol=symbol))
        return ValueResult(rows, make_record(SOURCE, "complete",
                                             collected_at=datetime.now().astimezone()))
    except Exception as error:  # noqa: BLE001 — 第三方估值历史边界
        logger.warning("东财估值历史失败 %s: %s", symbol, type(error).__name__)
        return ValueResult(None, make_record(
            SOURCE, "failed", error_kind=type(error).__name__,
            error_message="东财估值历史暂不可用"), [str(error)])
```

def _fetch_valuation(self, symbol: str, **kwargs) -> list[ValuationData]:
    """估值：腾讯快照为主，东财日频最新行和百度曲线为备源。"""
    if kwargs.get("market") == "hk":
        return self._fetch_hk_valuation(symbol)
    if kwargs.get("market") == "us":
        from data.us_stock import us_valuation
        return us_valuation(symbol)
    import requests as _req
    from data.a_valuation_history import load_history
    from utils.numbers import is_positive_finite

    tx_symbol = f"sh{symbol}" if symbol.startswith("6") else f"sz{symbol}"
    pe = pb = None
    pe_as_of = pb_as_of = None
    tencent_failed = False
    try:
        response = _req.get(f"https://qt.gtimg.cn/q={tx_symbol}", timeout=10)
        fields = response.text.split("~")
        if len(fields) < 47:
            raise ValueError(f"腾讯快照字段数异常: {len(fields)}")
        pe = float(fields[39]) if fields[39] else None
        pb = float(fields[46]) if fields[46] else None
        stamp = fields[30]
        if len(stamp) == 14 and stamp.isdigit():
            candidate = datetime.strptime(stamp, "%Y%m%d%H%M%S").replace(
                tzinfo=timezone(timedelta(hours=8))).date()
            if candidate <= datetime.now(timezone(timedelta(hours=8))).date():
                pe_as_of = candidate if pe is not None else None
                pb_as_of = candidate if pb is not None else None
    except Exception as error:  # 腾讯快照边界，继续尝试东财
        tencent_failed = True
        self._record_collection_error("腾讯估值快照", error)

    history_result = load_history(symbol)
    history = history_result.value or []
    latest = history[-1] if history else None
    notes: list[str] = []
    if tencent_failed:
        notes.append("腾讯估值快照失败，使用东财日频最新行兜底")
    if history_result.value is None:
        notes.append(f"东财日频估值历史不可用：{history_result.record.error_message or '未取得数据'}")
    if latest is not None:
        if pe is None and latest.pe_ttm is not None:
            pe, pe_as_of = latest.pe_ttm, latest.trade_date
        if pb is None and latest.pb is not None:
            pb, pb_as_of = latest.pb, latest.trade_date

    baidu_curves: dict[str, dict[date, float]] = {}
    if not history:
        for metric, field in (("市盈率(TTM)", "pe"), ("市净率", "pb")):
            try:
                points = {day: value for day, value in
                          baidu_metric(symbol, metric, market="ab").items()
                          if is_positive_finite(value)}
                if not points:
                    continue
                baidu_curves[field] = points
                day = max(points)
                if field == "pe" and pe is None:
                    pe, pe_as_of = points[day], day
                if field == "pb" and pb is None:
                    pb, pb_as_of = points[day], day
            except Exception as error:  # 百度网络边界
                self._record_collection_error(f"百度A股{metric}", error)
                notes.append(f"{metric} 百度曲线不可用")
    if pe is None and pb is None:
        return []

    cutoff = datetime.now().astimezone().date()
    extended_pe: dict[date, float] = {}
    extended_pb: dict[date, float] = {}
    pe_history: dict[date, float] = {}
    pb_history: dict[date, float] = {}
    if history:
        start_long = cutoff - timedelta(days=1827)

                       if start_long <= row.trade_date <= cutoff
                       and is_positive_finite(row.pe_ttm)}
        extended_pb = {row.trade_date: row.pb for row in history
                       if start_long <= row.trade_date <= cutoff
                       and is_positive_finite(row.pb)}
        short_start = cutoff - timedelta(days=365)
        pe_history = {day: value for day, value in extended_pe.items() if day >= short_start}
        pb_history = {day: value for day, value in extended_pb.items() if day >= short_start}
        notes.append("东财日频估值历史：2018 年至今，PE/PB/PS 分别标注日期；PS 仅展示")
        if latest is not None and latest.total_shares:
            notes.append(f"东财总股本：{latest.total_shares:.0f}")
    else:
        pe_history = baidu_curves.get("pe", {})
        pb_history = baidu_curves.get("pb", {})
        if pe_history or pb_history:
            notes.append("百度近一年历史：各指标按自身有效日期计算")

    known_dates = [day for day in (pe_as_of, pb_as_of, latest.trade_date if latest else None)
                   if day is not None]
    item = ValuationData(
        symbol=symbol,

    snapshot = ctx.valuation_data

    result: dict[str, dict] = {}
    for metric, curve in (("pe", snapshot.extended_pe_history),
                          ("pb", snapshot.extended_pb_history)):
        samples = sorted((day, value) for day, value in curve.items()
                         if start <= day <= cutoff and is_positive_finite(value))
        if not samples:
            continue
        values = [value for _, value in samples]
        result[metric] = {
            "window": snapshot.history_window_label,
            "source": source,
            "start": str(samples[0][0]),
            "end": str(samples[-1][0]),
            "sample_count": len(values),
            "reference": values[-1],
            "percentile": (round(sum(value < values[-1] for value in values)
                                 / len(values) * 100, 1) if len(values) >= 252 else None),
        }
    return result

```python
"""同日有效同行市值的部分覆盖排名；缺失同行计入覆盖缺口。"""
from typing import Any

from data.peer_comparison_rules import valuation_date
from utils.numbers import is_positive_finite


def market_cap_rank(snapshot: dict[str, Any], symbols: list[str], target: str) -> dict[str, Any]:
    expected = set(symbols) | {target}
    base = {"rank": None, "total": 0, "pool_total": len(expected),
            "coverage": f"0/{len(expected)}", "as_of": "", "reason": "完整同日市值覆盖不足"}
    target_row = snapshot.get("target") or {}
    if target_row.get("symbol") != target:
        return {**base, "reason": "目标记录缺失或身份不匹配"}
    by_code: dict[str, dict[str, Any]] = {target: target_row}
    for row in snapshot.get("peers", []):
        symbol = row.get("symbol")
        if symbol is None or symbol not in expected or symbol == target:
            return {**base, "reason": "同行数据含未知或重复代码"}
        if symbol in by_code:
            return {**base, "reason": "同行池存在重复代码"}
        by_code[symbol] = row
    target_date = valuation_date(target_row.get("as_of"))
    if not target_date or not is_positive_finite(target_row.get("market_cap")):
        return {**base, "reason": "目标市值缺失或日期无效"}
    if snapshot.get("as_of") and valuation_date(snapshot["as_of"]) != target_date:
        return {**base, "reason": "快照日期与目标日期不一致"}
    valid = [row for code, row in by_code.items()
             if code != target and valuation_date(row.get("as_of")) == target_date
             and is_positive_finite(row.get("market_cap"))]
    total = 1 + len(valid)
    if total == 1:
        return {**base, "reason": "没有同日有效同行市值"}
    rank = 1 + sum(row["market_cap"] > target_row["market_cap"] for row in valid)
    return {"rank": rank, "total": total, "pool_total": len(expected),
            "coverage": f"{total}/{len(expected)}", "as_of": target_date,
            "reason": "" if total == len(expected) else "部分同行缺少同日有效市值"}
```

"""港股逐期币种/单位交叉核验；不把目录币种当作金额证据。"""
import logging
import math
from dataclasses import dataclass
from datetime import date, datetime
from typing import Any

import akshare as ak

from data.hk_eastmoney import finite_number
from data.schemas import FinancialData

logger = logging.getLogger(__name__)
SOURCE = "东方财富港股分析指标"
_STANDARD = {"revenue": ("004001001", "004001002"), "profit": ("004025002",)}
_BANK = {"revenue": ("001003999",), "profit": ("001025002",)}


@dataclass(frozen=True)
class PeriodCurrencyEvidence:
    period: date
    start: date | None
    currency: str
    verified: bool
    reason: str


def _date(value: Any) -> date | None:
    try:
        return datetime.fromisoformat(str(value)[:10]).date()
    except (TypeError, ValueError):
        return None


def parse_indicator_rows(frame: Any) -> dict[date, dict[str, Any]]:
    if frame is None or getattr(frame, "empty", True):
        raise ValueError("港股分析指标为空")
    rows: dict[date, dict[str, Any]] = {}
    conflicts: set[date] = set()
    for record in frame.to_dict("records"):
        period = _date(record.get("REPORT_DATE"))
        if period is None:
            continue
        if period in rows and rows[period] != record:
            conflicts.add(period)
        rows[period] = record
    for period in conflicts:
        rows.pop(period, None)
    if not rows:
        raise ValueError("港股分析指标没有有效报告期")
    return rows


def load_indicator_periods(code: str) -> dict[date, dict[str, Any]]:
    return parse_indicator_rows(ak.stock_financial_hk_analysis_indicator_em(
        symbol=code, indicator="报告期"))


def _close(left: Any, right: Any) -> bool:
    a, b = finite_number(left), finite_number(right)
    return a is not None and b is not None and math.isclose(a, b, rel_tol=1e-6, abs_tol=1.0)


def verify_period(period: date, start: date | None,
                  income_rows: list[dict[str, Any]], indicator: dict[str, Any] | None,
                  *, bank: bool) -> PeriodCurrencyEvidence:
    if indicator is None:
        return PeriodCurrencyEvidence(period, start, "", False, "未取得同期分析指标")
    currency = FinancialData.normalize_report_currency(indicator.get("CURRENCY"))
    indicator_start = _date(indicator.get("START_DATE"))
    if not currency:
        return PeriodCurrencyEvidence(period, start, "", False, "分析指标未提供有效币种")
    if start is None or indicator_start != start:
        return PeriodCurrencyEvidence(period, start, currency, False, "分析指标与报表期间起点不一致")
    flag = str(indicator.get("IS_CNY_CODE") or "")
    if flag and ((currency == "CNY") != (flag == "1")):
        return PeriodCurrencyEvidence(period, start, currency, False, "IS_CNY_CODE 与币种矛盾")
    codes = _BANK if bank else _STANDARD
    amounts = {str(row.get("STD_ITEM_CODE")): finite_number(row.get("AMOUNT"))
               for row in income_rows if str(row.get("SECUCODE", "")).endswith(".HK")
               and _date(row.get("REPORT_DATE")) == period and _date(row.get("START_DATE")) == start}
    revenue = sum(amounts.get(code, 0.0) or 0.0 for code in codes["revenue"])
    profit = sum(amounts.get(code, 0.0) or 0.0 for code in codes["profit"])
    if not _close(revenue, indicator.get("OPERATE_INCOME")):
        return PeriodCurrencyEvidence(period, start, currency, False, "报表收入聚合与指标接口不一致")
    if not _close(profit, indicator.get("HOLDER_PROFIT")):
        return PeriodCurrencyEvidence(period, start, currency, False, "报表利润聚合与指标接口不一致")
    return PeriodCurrencyEvidence(period, start, currency, True, "")

def _hk_verified(rows: list[FinancialData]) -> bool:
    return bool(rows) and all(
        row.amount_currency_verified and row.report_currency for row in rows
    ) and len({row.report_currency for row in rows}) == 1


def financial_ttm(rows: list[FinancialData], field: str, market: str, *,
                  require_precision: bool = False) -> float | None:
    ordered = sorted(rows, key=lambda row: row.fiscal_quarter)
    if not ordered:
        return None
    latest = ordered[-1]
    from data.financial_metrics import native_amounts

    def usable(selected: list[FinancialData]) -> bool:
        return not require_precision or native_amounts(selected, (field,))
    if market == "us":
        recent = ordered[-4:]
        values = [getattr(row, field) for row in recent]
        return sum(values) if len(recent) == 4 and usable(recent) and all(is_finite_number(value) for value in values) else None
    end = latest.fiscal_quarter
    if end.month == 12:
        if market == "hk" and not _hk_verified([latest]):
            return None
        return getattr(latest, field) if usable([latest]) else None
    annual = next((row for row in reversed(ordered)
                   if row.fiscal_quarter.year == end.year - 1 and row.fiscal_quarter.month == 12), None)
    previous = next((row for row in reversed(ordered)
                     if row.fiscal_quarter.year == end.year - 1 and row.fiscal_quarter.month == end.month), None)
    if annual is None or previous is None:
        return None
    selected = [latest, annual, previous]
    if market == "hk" and not _hk_verified(selected):
        return None
    values = [getattr(row, field) for row in selected]
    return values[0] + values[1] - values[2] if usable(selected) and all(is_finite_number(value) for value in values) else None

```python
"""SEC XBRL 营业收入事实；只服务 BRK.B 展示，不进入通用评分。"""
import logging
from dataclasses import dataclass
from datetime import date, timedelta
from typing import Any

import requests

logger = logging.getLogger(__name__)
SOURCE = "SEC XBRL"
BRK_CIK = "0001067983"
BRK_CONCEPT = "RevenueFromContractWithCustomerExcludingAssessedTax"
USER_AGENT = "stock-robot research local@example.com"


@dataclass(frozen=True)
class SecRevenueFact:
    start: date
    end: date
    value: float
    form: str
    filed: date
    concept: str
    currency: str
    basis: str


def _date(value: Any) -> date | None:
    try:
        return date.fromisoformat(str(value)[:10])
    except (TypeError, ValueError):
        return None


def parse_company_concept(payload: dict[str, Any], concept: str) -> list[SecRevenueFact]:
    units = payload.get("units") or {}
    rows = units.get("USD")
    if not isinstance(rows, list):
        raise ValueError("SEC 响应缺少 USD 单位数据")
    facts: dict[tuple[date, date], SecRevenueFact] = {}
    for row in rows:
        if not isinstance(row, dict) or row.get("form") not in {"10-Q", "10-K"}:
            continue
        start, end, filed = _date(row.get("start")), _date(row.get("end")), _date(row.get("filed"))
        value = row.get("val")
        if start is None or end is None or filed is None or start > end:
            continue
        if not isinstance(value, (int, float)) or isinstance(value, bool) or value <= 0:
            continue
        fact = SecRevenueFact(start, end, float(value), str(row["form"]), filed,
                              concept, "USD", "直接披露")
        previous = facts.get((start, end))
        if previous is None or filed >= previous.filed:
            facts[(start, end)] = fact
    if not facts:
        raise ValueError("SEC 营业收入没有有效期间")
    return [facts[key] for key in sorted(facts)]


def single_quarters(facts: list[SecRevenueFact]) -> list[SecRevenueFact]:
    output: dict[date, SecRevenueFact] = {}
    for fact in facts:
        days = (fact.end - fact.start).days
        if 80 <= days <= 100:
            output[fact.end] = fact
            continue
        if 170 <= days <= 190:
            q1 = next((item for item in facts if item.start == fact.start and 80 <= (item.end - item.start).days <= 100), None)
            if q1 is not None and fact.value > q1.value:
                output[fact.end] = SecRevenueFact(
                    q1.end + timedelta(days=1), fact.end, fact.value - q1.value,
                    fact.form, fact.filed, fact.concept, fact.currency, "累计差额")
            continue
        if 260 <= days <= 280:
            q1 = next((item for item in facts if item.start == fact.start and 80 <= (item.end - item.start).days <= 100), None)
            h1 = next((item for item in facts if item.start == fact.start and 170 <= (item.end - item.start).days <= 190), None)
            if q1 is not None and h1 is not None and fact.value > h1.value:
                output[fact.end] = SecRevenueFact(
                    h1.end + timedelta(days=1), fact.end, fact.value - h1.value,
                    fact.form, fact.filed, fact.concept, fact.currency, "累计差额")
            continue
        if 350 <= days <= 380:
            q3 = next((item for item in facts if item.start == fact.start and 260 <= (item.end - item.start).days <= 280), None)
            if q3 is not None and fact.value > q3.value:
                output[fact.end] = SecRevenueFact(
                    q3.end + timedelta(days=1), fact.end, fact.value - q3.value,
                    fact.form, fact.filed, fact.concept, fact.currency, "累计差额")
    return [output[key] for key in sorted(output)]


def load_brk_revenue() -> list[SecRevenueFact]:
    url = (f"https://data.sec.gov/api/xbrl/companyconcept/CIK{BRK_CIK}/us-gaap/{BRK_CONCEPT}.json")
    response = requests.get(url, headers={"User-Agent": USER_AGENT}, timeout=20)
    response.raise_for_status()
    facts = parse_company_concept(response.json(), BRK_CONCEPT)
    return single_quarters(facts)
```
