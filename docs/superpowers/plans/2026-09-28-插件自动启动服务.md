# 插件自动启动服务 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** AstrBot 插件在启用时自动把 stock_robot 分析服务准备好（复用已在跑的实例，否则下载源码、建独立环境、装依赖、写配置、启动），并把主仓的 RAG 重依赖拆成 `rag` extra。

**Architecture:** 插件侧新增 `bootstrap.py`（下载/环境/配置四步纯函数 + 命令执行）与 `launcher.py`（健康探测 → 自举 → 启动 → 只结束自己拉起的进程），`main.py` 在 `initialize()`/工具调用/`terminate()` 三处挂钩；`client.py` 不改。自举实例固定在 `{AstrBot plugin_data}/astrbot_plugin_stock_robot/instance/`，`cwd=instance` 决定主仓的 `.stock_robot/` 与 `reports/` 落点。主仓把 `chromadb` + `sentence-transformers` 移入 `[project.optional-dependencies] rag`。

**Tech Stack:** Python 3.11+ / httpx（下载与探测）/ zipfile（解压）/ asyncio（子进程与单飞）/ PyYAML（写配置）/ pytest + respx（插件仓单测）/ click + pytest（主仓）。

**Spec:** `docs/superpowers/specs/2026-09-28-插件自动启动服务设计.md`

**两个仓库：**
- 主仓 `D:\code\stock_robot`（python 一律用 `.venv/Scripts/python`；校验命令见 `AGENTS.md`）
- 插件仓 `D:\code\astrbot_plugin_stock_robot`（python 一律用 `.venv/Scripts/python.exe`）

```python
"""自举 stock_robot 实例：下载源码、建独立环境、装依赖、写配置。

不依赖 astrbot，可独立单测；跨平台差异集中在本模块的路径/命令辅助函数里。
"""

from __future__ import annotations

import asyncio
import io
import logging
import os
import shutil
import subprocess
import sys
import zipfile
from collections.abc import Awaitable, Callable
from dataclasses import dataclass
from pathlib import Path

import httpx
import yaml

logger = logging.getLogger(__name__)

DEFAULT_ARCHIVE_URL = (
    "https://github.com/Arrhenius401/stock_robot/archive/refs/heads/main.zip"
)
DOWNLOAD_TIMEOUT_SECONDS = 300
OUTPUT_TAIL_LINES = 5

Runner = Callable[[list[str], Path], Awaitable[tuple[int, str]]]


@dataclass(frozen=True)
class StepResult:
    """自举单步结果；失败时 detail 是可写入日志与用户消息的原因摘要。"""

    ok: bool
    detail: str = ""


def venv_dir(instance: Path) -> Path:
    return instance / "venv"


def venv_python(instance: Path) -> Path:
    """venv 内解释器：Windows 用 Scripts，POSIX 用 bin。"""
    name = "Scripts/python.exe" if os.name == "nt" else "bin/python"
    return venv_dir(instance) / name


def venv_launcher(instance: Path) -> Path:
    """venv 内 stock-robot 入口脚本。"""
    name = "Scripts/stock-robot.exe" if os.name == "nt" else "bin/stock-robot"
    return venv_dir(instance) / name


def spawn_kwargs() -> dict:
    """子进程创建参数；Windows 下抑制控制台窗口。"""
    if os.name == "nt":
        return {"creationflags": subprocess.CREATE_NO_WINDOW}
    return {}


async def run_command(cmd: list[str], cwd: Path) -> tuple[int, str]:
    """执行外部命令并合并标准输出/错误；返回 (退出码, 输出文本)。"""
    proc = await asyncio.create_subprocess_exec(
        *cmd,
        cwd=str(cwd),
        stdout=asyncio.subprocess.PIPE,
        stderr=asyncio.subprocess.STDOUT,
        **spawn_kwargs(),
    )
    out, _ = await proc.communicate()
    return proc.returncode or 0, out.decode("utf-8", "replace")


def _tail(prefix: str, output: str) -> str:
    lines = [line.strip() for line in output.strip().splitlines() if line.strip()]
    return f"{prefix}：{' / '.join(lines[-OUTPUT_TAIL_LINES:])}"


def extract_archive(data: bytes, src_dir: Path) -> StepResult:
    """解压归档到 src_dir：剥离压缩包自带的顶层目录，拒绝越界成员。"""
    src_dir.mkdir(parents=True, exist_ok=True)
    root = src_dir.resolve()
    try:
        with zipfile.ZipFile(io.BytesIO(data)) as zf:
            names = [n for n in zf.namelist() if not n.endswith("/")]
            if not names:
                return StepResult(False, "压缩包内没有文件")
            tops = {n.split("/", 1)[0] for n in names}
            strip = tops.pop() if len(tops) == 1 else ""
            for name in names:
                relative = name.split("/", 1)[1] if strip and "/" in name else name
                if not relative:
                    continue
                target = (src_dir / relative).resolve()
                if not target.is_relative_to(root):
                    return StepResult(False, f"压缩包成员路径越界：{name}")
                target.parent.mkdir(parents=True, exist_ok=True)
                target.write_bytes(zf.read(name))
    except zipfile.BadZipFile as exc:
        return StepResult(False, f"压缩包无法解析：{exc}")
    return StepResult(True)
```

```python
    )
    code, output = await runner(cmd, src_dir)

        return StepResult(False, _tail("安装依赖失败", output))
```

```python
"""服务就绪编排：复用已在跑的服务 → 自举并启动 → 只结束自己拉起的进程。

不依赖 astrbot，可独立单测。
"""

from __future__ import annotations

import asyncio
import logging
from collections.abc import Awaitable, Callable
from dataclasses import dataclass
from pathlib import Path
from typing import IO, Literal

import httpx

from bootstrap import Runner, ensure_instance, run_command, spawn_kwargs, venv_launcher

logger = logging.getLogger(__name__)

HEALTH_TIMEOUT_SECONDS = 2.0
READY_POLL_INTERVAL_SECONDS = 1.0
STOP_GRACE_SECONDS = 5.0
LOCAL_HOSTS = {"127.0.0.1", "localhost", "::1"}

State = Literal["idle", "starting", "ready", "failed"]


@dataclass(frozen=True)
class ReadyOutcome:
    ok: bool
    reason: str = ""


async def probe_health(base_url: str, timeout: float = HEALTH_TIMEOUT_SECONDS) -> bool:
    """GET {base_url}/health 是否返回 200。"""
    try:
        async with httpx.AsyncClient(timeout=timeout) as client:
            resp = await client.get(f"{base_url.rstrip('/')}/health")
    except httpx.HTTPError:
        return False
    return resp.status_code == 200


class ServiceLauncher:
    """解析服务来源并保证就绪；并发调用共享同一次启动（单飞）。"""

    def __init__(
        self,
        base_url: str,
        port: int,
        archive_url: str,
        extras: str,
        auto_install: bool,
        timeout: float,
        data_dir: Path,
        llm: dict | None = None,
        *,
        runner: Runner = run_command,
        command_prefix: list[str] | None = None,
    ):
        self._base_url = base_url.rstrip("/")
        self._port = port
        self._archive_url = archive_url
        self._extras = extras
        self._auto_install = auto_install
        self._timeout = timeout
        self._data_dir = data_dir
        self._llm = llm
        self._runner = runner
        self._command_prefix = command_prefix
        self._instance = data_dir / "instance"
        self._lock = asyncio.Lock()
        self._process: asyncio.subprocess.Process | None = None
        self._log_handle: IO[bytes] | None = None
        self._stopped = False
        self._state: State = "idle"
        self._reason = ""

    @property
    def state(self) -> State:
        return self._state

    @property
    def owns_process(self) -> bool:
        """当前是否持有一个由本实例启动的进程。"""
        return self._process is not None and self._process.returncode is None

    async def ensure_ready(self, allow_install: bool) -> ReadyOutcome:
        """保证服务就绪；allow_install=False 时不下载不安装，只复用或启动已有实例。"""
        if self._state == "ready" and await probe_health(self._base_url):
            return ReadyOutcome(True)
        async with self._lock:
            return await self._ensure_ready_locked(allow_install)

    async def stop(self) -> None:
        """结束由本实例拉起的进程；用户自己的实例不受影响。"""
        self._stopped = True
        proc = self._process
        self._process = None
        if proc is not None and proc.returncode is None:
            proc.terminate()
            try:
                await asyncio.wait_for(proc.wait(), timeout=STOP_GRACE_SECONDS)
            except asyncio.TimeoutError:
                proc.kill()
                await proc.wait()
        self._close_log()
        self._state = "idle"

    # ------------------------------------------------------------------
    # 内部

    async def _ensure_ready_locked(self, allow_install: bool) -> ReadyOutcome:
        if await probe_health(self._base_url):
            self._state = "ready"
            return ReadyOutcome(True)
        if not self._is_local():
            return self._fail(f"{self._base_url} 不是本机地址，插件无法启动服务")
        if not venv_launcher(self._instance).exists():
            if not (allow_install and self._auto_install):
                return self._fail("分析服务尚未安装（可能正在安装中），详见插件日志")
            self._state = "starting"
            logger.info("开始自举 stock_robot 到 %s", self._instance)
            result = await ensure_instance(
                self._instance, self._archive_url, self._extras, self._llm, self._port,
                runner=self._runner,
            )
            if not result.ok:
                return self._fail(result.detail)
        return await self._start()

    def _is_local(self) -> bool:
        return httpx.URL(self._base_url).host in LOCAL_HOSTS

    def _command(self) -> list[str]:
        prefix = self._command_prefix or [str(venv_launcher(self._instance))]
        return [*prefix, "run", "--host", httpx.URL(self._base_url).host, "--port", str(self._port)]

    async def _start(self) -> ReadyOutcome:
        if self._stopped:
            return self._fail("插件已停用，未启动服务")
        self._state = "starting"
        log_path = self._data_dir / "service.log"
        log_path.parent.mkdir(parents=True, exist_ok=True)
        self._log_handle = log_path.open("ab")
        self._process = await asyncio.create_subprocess_exec(
            *self._command(),
            cwd=str(self._instance),
            stdout=self._log_handle,
            stderr=asyncio.subprocess.STDOUT,
            **spawn_kwargs(),
        )
        logger.info("已启动 stock_robot 进程：%s", " ".join(self._command()))
        deadline = asyncio.get_running_loop().time() + self._timeout
        while asyncio.get_running_loop().time() < deadline:
            if self._process.returncode is not None:
                return self._fail(
                    f"服务进程退出（码 {self._process.returncode}）：{self._log_tail(log_path)}"
                )
            if await probe_health(self._base_url):
                self._state = "ready"
                return ReadyOutcome(True)
            await asyncio.sleep(READY_POLL_INTERVAL_SECONDS)
        return self._fail(f"服务启动超时（{int(self._timeout)} 秒），详见 {log_path}")

    @staticmethod
    def _log_tail(log_path: Path) -> str:
        try:
            lines = log_path.read_text("utf-8", errors="replace").strip().splitlines()
        except OSError:
            return "（无日志）"
        return lines[-1] if lines else "（无日志）"

    def _close_log(self) -> None:
        if self._log_handle is not None:
            self._log_handle.close()
            self._log_handle = None

    def _fail(self, reason: str) -> ReadyOutcome:
        self._state = "failed"
        self._reason = reason
        logger.warning("分析服务未就绪：%s", reason)
        self._close_log()
        return ReadyOutcome(False, reason)
```

```python
def read_llm_config(context: Context) -> dict | None:
    """把 AstrBot 当前模型供应商转成主仓 llm 配置；读不到返回 None。"""

    keys = config.get("key") or []
    api_key = str(keys[0]) if keys else ""
    if not api_key:

    provider_type = str(config.get("type") or "")

        "provider": "claude" if ("anthropic" in provider_type or "claude" in provider_type) else "openai",
```
